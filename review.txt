    47export function computeFilteredUnits(units, cellsByUnitStage, stages, filters) {
    48  if (!Array.isArray(units)) return [];
    49  const f = filters || initialFilters();
    50  const stageList = Array.isArray(stages) ? stages : [];
    51  const cells = cellsByUnitStage || {};
    52
    53  return units.filter(u => {
    54    // 1. Building (OR within, AND with rest)
    55    if (f.building_ids?.length > 0 && !f.building_ids.includes(u.building_id)) {
    56      return false;
    57    }
    58
    59    // 2. Apartment search (substring on unit_no)
    60    if (f.apartment_search) {
    61      const q = String(f.apartment_search).toLowerCase();
    62      if (!String(u.unit_no ?? '').toLowerCase().includes(q)) return false;
    63    }
    64
    65    // 3. Per-stage status filters
    66    for (const [stageId, allowed] of Object.entries(f.stage_status_filters || {})) {
    67      if (!allowed?.length) continue;
    68      const cell = cells[`${u.id}::${stageId}`];
    69      const status = cell?.status ?? EMPTY;
    70      if (!allowed.includes(status)) return false;
    71    }
    72
    73    // 4. Per-tag value filters
    74    for (const [stageId, allowed] of Object.entries(f.tag_value_filters || {})) {
    75      if (!allowed?.length) continue;
    76      const cell = cells[`${u.id}::${stageId}`];
    77      const raw = cell?.text_value;
    78      const value = (raw == null || raw === '') ? EMPTY : raw;
    79      if (!allowed.includes(value)) return false;
    80    }
    81
    82    // 5. Global free-text search across unit_no + cell text + cell note
    83    if (f.search_text) {
    84      const q = String(f.search_text).toLowerCase();
    85      const inUnitNo = String(u.unit_no ?? '').toLowerCase().includes(q);
    86      const inCells = stageList.some(s => {
    87        const c = cells[`${u.id}::${s.id}`];
    88        return (
    89          String(c?.text_value ?? '').toLowerCase().includes(q) ||
    90          String(c?.note ?? '').toLowerCase().includes(q)
    91        );
    92      });
    93      if (!inUnitNo && !inCells) return false;
    94    }
    95
    96    return true;
    97  });
    98}

   101export function computeActiveCount(filters) {
   102  const f = filters || initialFilters();
   103  const stage_status = {};
   104  let stageStatusTotal = 0;
   105  for (const [k, v] of Object.entries(f.stage_status_filters || {})) {
   106    if (v?.length) { stage_status[k] = v.length; stageStatusTotal += v.length; }
   107  }
   108  const tag_value = {};
   109  let tagValueTotal = 0;
   110  for (const [k, v] of Object.entries(f.tag_value_filters || {})) {
   111    if (v?.length) { tag_value[k] = v.length; tagValueTotal += v.length; }
   112  }
   113  const building = f.building_ids?.length || 0;
   114  const apartment = f.apartment_search ? 1 : 0;
   115  const search = f.search_text ? 1 : 0;
   116  return {
   117    total: building + apartment + search + stageStatusTotal + tagValueTotal,
   118    building, apartment, search,
   119    stage_status, tag_value,
   120  };

   143    for (const [k, v] of Object.entries(dict || {})) {
